--- MULTI-SOURCE METADATA LOOKUP --- def fetch_metadata(self, isbn): print(f"\n--- [LOOKUP START: ISBN {isbn}] ---") title = "Unknown Title" author = "Unknown Author" genre = "General" format_info = "Physical Edition" cover_url = "" # 1. Google Books (Exact) try: url = f"https://www.googleapis.com/books/v1/volumes?q=isbn:{isbn}" resp = requests.get(url, headers=HEADERS, timeout=4) if resp.status_code == 200: data = resp.json() if data.get("totalItems", 0) > 0: v = data["items"][0]["volumeInfo"] title = v.get("title", title) if "authors" in v: author = ", ".join(v["authors"]) if "categories" in v and len(v["categories"]) > 0: genre = v["categories"][0].split(" / ")[0] if "pageCount" in v: format_info = f"Pages: {v['pageCount']}" if "imageLinks" in v: cover_url = v["imageLinks"].get("thumbnail", "").replace("http://", "https://") print(f"-> Google Books Exact: {title} by {author}") except Exception as e: print(f"Google Books exact error: {e}") # 2.
